\section{Espacios simplemente conexos, deformación de aplicaciones y grupo fundamental de $S^n$}

\subsection{Espacios simplemente conexos}

\begin{observacion}{}{}
Por la \cref{prop:equiv_homotopica}, si $X$ es contráctil (es decir, $X \simeq \{p\}$), entonces $\pi_1(X) = 1$.
\end{observacion}

\begin{definicion}{Espacio simplemente conexo [C7.1]}{def_simp_conexo}
Sea $X$ un espacio topológico. Diremos que $X$ es simplemente conexo cuando es arco-conexo y $\pi_1(X) = 1$.
\end{definicion}

\begin{proposicion}{ [C7.2]}{prop_arco_conexo}
Sea $X$ un espacio arco-conexo. Entonces:

$$\pi_1(X) = 1 \iff \forall x_0, x_1 \in X, \forall \alpha, \beta \in C_{x_0, x_1}(X) : \alpha \simeq_{\{0,1\}} \beta$$
\end{proposicion}

//Nota para cuando tenga creditos de hacer grafico de espacio con agujero y ver como ahi no se pueden deformar las cosas

\begin{proof}
$\Rightarrow)$ Sean $x_0, x_1 \in X$ y sean $\alpha, \beta \in C_{x_0, x_1}(X)$.

Como $X$ es arco-conexo y $\pi_1(X) \cong \pi_1(X, x_0) = 1 = [c_{x_0}]$, consideramos el lazo $f := \alpha * \overline{\beta} \in C_{x_0, x_0}(X)$.
Dado que $\pi_1(X) = 1$, se cumple que $[\alpha * \overline{\beta}] = 1$. Esto implica que $[\alpha] \cdot [\beta^{-1}] = 1$, y por tanto $[\alpha] = [c_{x_0} * \beta] = [\beta]$ y por consiguiente $\alpha \simeq_{\{0,1\}} \beta$.

$\Leftarrow)$ Trivial tomando $x_0 = x_1$.
\end{proof}

\subsection{Reparametrización de arcos y lema de Lebesgue}

\begin{proposicion}{ [C7.3]}{prop_reparametrizacion}
Sea $g: I \longrightarrow I$ una aplicación continua tal que $g(0)=0$ y $g(1)=1$. Sea $\alpha: I \longrightarrow X$ una aplicación continua, siendo $X$ un espacio topológico cualquiera. Entonces se cumple que $\alpha \simeq_{\{0,1\}} \alpha \circ g$.
\end{proposicion}
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.5]
  \draw[->] (-0.2,0) -- (1.5,0);
  \draw[->] (0,-0.2) -- (0,1.5);
  \draw (1,0.05) -- (1,-0.05) node[below] {$1$};
  \draw (0.05,1) -- (-0.05,1) node[left] {$1$};
  \draw[blue, thick] (0,0) .. controls (0.2,0.8) and (0.6,0.3) .. (1,1);
\end{tikzpicture}
\caption{Gráfica de la reparametrización continua $g: I \longrightarrow I$.}
\end{figure}
\begin{proof}
Sea $F: I \times I \longrightarrow X$ la aplicación continua definida por $F(s,t) = \alpha((1-t)s + t g(s))$. Comprobamos sus valores:
\begin{itemize}
\item $F_0(s) = F(s,0) = \alpha((1-0)s + 0 \cdot g(s)) = \alpha(s)$, $\forall s \in I$.
\item $F_1(s) = F(s,1) = \alpha((1-1)s + 1 \cdot g(s)) = \alpha(g(s))$, $\forall s \in I$.
\item $F(0,t) = \alpha((1-t)0 + t g(0)) = \alpha(0)$, que no depende de $t$.
\item $F(1,t) = \alpha((1-t)1 + t g(1)) = \alpha((1-t) + t) = \alpha(1)$, que tampoco depende de $t$.
\end{itemize}
\end{proof}

\begin{teorema}{Lema de Lebesgue [C7.4]}{lema_lebesgue}
Sea $(X,d)$ un espacio métrico compacto y sea $\mathcal{U} = \{U_i\}_{i \in \Lambda}$ un recubrimiento abierto de $X$ (lo que significa que $U_i$ es abierto en $X$ y además $\bigcup_{i \in \Lambda} U_i = X$). Entonces, $\exists \delta > 0$ tal que $\forall x \in X, \exists i(x) \in \Lambda$ verificando que $B(x,\delta) \subseteq U_{i(x)}$.
\end{teorema}

\begin{proof}
Dado que $\bigcup_{i \in \Lambda} U_i = X$, se tiene que $\forall x \in X, \exists r(x) > 0$ tal que $B(x, r(x)) \subseteq U_{i(x)}$ para algún $i(x) \in \Lambda$.
Definimos la familia $\mathcal{F} = \{B(x, \frac{r(x)}{2})\}_{x \in X}$. Claramente $\mathcal{F}$ es un recubrimiento abierto de $X$. Como $X$ es compacto, admite un subrecubrimiento finito de la forma:

$$X = B\left(x_1, \frac{r(x_1)}{2}\right) \cup \dots \cup B\left(x_n, \frac{r(x_n)}{2}\right)$$
.
Sea $\delta = \min \left\{ \frac{r(x_1)}{2}, \dots, \frac{r(x_n)}{2} \right\}$.
Sea $x \in X$. Existe un índice $j \in \{1, \dots, n\}$ tal que $x \in B(x_j, \frac{r(x_j)}{2})$. Veamos que $B(x,\delta) \subseteq U_{i(x_j)}$:
Si $y \in B(x,\delta)$, entonces

$$d(y, x_j) \le d(y,x) + d(x,x_j) < \delta + \frac{r(x_j)}{2} \le \frac{r(x_j)}{2} + \frac{r(x_j)}{2} = r(x_j)$$
.
Hemos llegado a la conclusión de que $B(x,\delta) \subseteq B(x_j, r(x_j)) \subseteq U_{i(x_j)}$.
\end{proof}

\begin{proposicion}{Lebesgue para recubrimientos de $[0,1]$ [C7.5]}{cor_lebesgue_I}
Sea $\mathcal{U} = \{U_i\}_{i \in \Lambda}$ un recubrimiento abierto de $I = [0,1]$. Entonces existe $n \geq 1$ tal que, para todo $j \in \{1,\dots,n\}$, existe $i(j) \in \Lambda$ con $\left[\frac{j-1}{n}, \frac{j}{n}\right] \subseteq U_{i(j)}$. Es decir, la partición $t_j = \frac{j}{n}$ (con distancia constante $\frac1n$ entre puntos consecutivos) subordinada a $\mathcal{U}$ puede tomarse \textbf{regular}.
\end{proposicion}

\begin{proof}
Por el lema de Lebesgue existe $\delta > 0$ tal que, para todo $x \in I$, existe $i(x) \in \Lambda$ con $\left]x-\delta, x+\delta\right[ \cap I \subseteq U_{i(x)}$. Sea $n \in \mathbb{Z}_{\geq 1}$ con $\frac1n < \delta$ y sea $t_j = \frac jn$, para $j = 0,\dots,n$. Fijado $j \in \{1,\dots,n\}$, sea $x_j$ el punto medio de $[t_{j-1},t_j]$ y escribamos $i(j) := i(x_j)$. Entonces
\[ [t_{j-1},t_j] \subseteq \left]x_j - \tfrac1n, x_j + \tfrac1n\right[ \cap I \subseteq \left]x_j - \delta, x_j + \delta\right[ \cap I \subseteq U_{i(j)}. \]
\end{proof}

\subsection{Condición suficiente para ser simplemente conexo}

\begin{teorema}{Condición suficiente para que un espacio sea simplemente conexo [C7.6]}{cond_simp_conexo}
Sea $X$ un espacio topológico y sea $\{V_i\}_{i \in \Lambda}$ un recubrimiento abierto de $X$. Supongamos que:
\begin{itemize}
\item[a)] $V_i$ es simplemente conexo para todo $i \in \Lambda$.
\item[b)] Para todo $i \neq j$, se cumple que $V_i \cap V_j$ es arco-conexo.
\item[c)] $\bigcap_{i \in \Lambda} V_i \neq \emptyset$.
\end{itemize}
Entonces, $X$ es simplemente conexo. 
\end{teorema}

\begin{proof}
$X$ es arco-conexo: está recubierto por los $V_i$, que son arco-conexos por ser simplemente conexos (hipótesis a), y su intersección no es vacía (hipótesis c), así que dos puntos cualesquiera se unen pasando por un punto común.

Sea $x_0 \in \bigcap_{i} V_i$. Basta ver que $\pi_1(X,x_0) = 1$. Sea $\alpha \in C_{x_0}(X)$; hay que probar que $\alpha \simeq_{\{0,1\}} c_{x_0}$.

Pasamos el recubrimiento de $X$ a $I$ (tomando antiimágenes por $\alpha$), de forma que $\alpha$ lleve cada trozo de $I$ dentro de un único $V_j$.
\begin{itemize}
    \item[] $\mathcal{U} = \{\alpha^{-1}(V_i)\}_{i \in \Lambda}$ son abiertos de $I$ (pues $\alpha$ es continua y los $V_i$ abiertos) y recubren $I$ (pues los $V_i$ recubren $X$), luego es recubrimiento abierto y podemos aplicar Lebesgue.
    \item[] Por Lebesgue en $[0,1]$ (\cref{prop:cor_lebesgue_I}) existe $n \geq 1$ tal que, con $t_j = \frac{j}{n}$, cada $[t_{j-1},t_j] \subseteq \alpha^{-1}(V_i)$ para algún $i$. Llamando $V_j$ a ese $V_i$, para cada $j = 1,\dots,n$ se tiene $\alpha([t_{j-1},t_j]) \subseteq V_j$.
    \item[] $t_j$ está en $[t_{j-1},t_j]$ y en $[t_j,t_{j+1}]$, luego $\alpha(t_j) \in V_j \cap V_{j+1}$ ($j = 1,\dots,n-1$). Además $x_0 \in V_j \cap V_{j+1}$ para cada $j = 1,\dots,n-1$, por elección de $x_0$.
\end{itemize}

Ahora parametrizamos $\alpha$ según los trozos que caben en cada $V_j$. Para $j = 1,\dots,n$ definimos
\[ \alpha_j \colon I \to V_j, \qquad \alpha_j(s) = \alpha\left(t_{j-1} + \tfrac{s}{n}\right). \]
Cuando $s$ va de $0$ a $1$, el punto $t_{j-1} + \frac{s}{n}$ recorre linealmente $[t_{j-1},t_j]$, así que $\alpha_j$ es el tramo $j$-ésimo de $\alpha$ reparametrizado a $I$, y su imagen está en $V_j$ por la elección de $V_j$. Va de $\alpha(t_{j-1})$ a $\alpha(t_j)$. Además $\alpha = \alpha_1 * \dots * \alpha_n$ como aplicaciones. En efecto, por definición de concatenación conjunta (\cref{def:concat_conjunta}), en el $j$-ésimo tramo se recorre $\alpha_j$:
\[ (\alpha_1 * \dots * \alpha_n)(u) = \alpha_j\big(nu - (j-1)\big) \qquad \text{si } u \in [t_{j-1},t_j]. \]
Sustituyendo la definición de $\alpha_j$ con $s = nu - (j-1) \in [0,1]$:
\[ \alpha_j\big(nu - (j-1)\big) = \alpha\Big(\tfrac{j-1}{n} + \tfrac{nu - (j-1)}{n}\Big) = \alpha(u). \]
Es decir, la concatenación comprime cada tramo en el mismo factor $n$ en que $\alpha_j$ lo había estirado, y se recupera $\alpha$.

Unimos cada punto de corte $\alpha(t_j)$ con $x_0$ mediante un camino $\rho_j$ dentro de $V_j \cap V_{j+1}$: así cada trozo se completa a un lazo en $x_0$ contenido en un único $V_j$ y, como $V_j$ es simplemente conexo, ese lazo tiene la clase del lazo constante $[c_{x_0}]$.
Sea $j \in \{1,\dots,n-1\}$. Como vimos al trocear $I$, $x_0$ y $\alpha(t_j)$ están en $V_j \cap V_{j+1}$, que es arco-conexo (hipótesis b; si $V_j = V_{j+1}$, es $V_j$, arco-conexo por a). Luego existe
\[ \rho_j \colon I \to V_j \cap V_{j+1}, \qquad \rho_j(0) = x_0, \quad \rho_j(1) = \alpha(t_j). \]

Como $\rho_j^{-1} * \rho_j \simeq_{\{0,1\}} c_{\alpha(t_j)}$, insertarlo entre $\alpha_j$ y $\alpha_{j+1}$ no cambia la clase:
\begin{align*}
[\alpha] &= [\alpha_1 * \alpha_2 * \dots * \alpha_n] \\
&= [\alpha_1 * \rho_1^{-1} * \rho_1 * \alpha_2 * \rho_2^{-1} * \rho_2 * \dots * \rho_{n-1}^{-1} * \rho_{n-1} * \alpha_n] \\
&= [\beta_1] \cdot [\beta_2] \cdots [\beta_n],
\end{align*}
con $\beta_1 = \alpha_1 * \rho_1^{-1}$, $\beta_j = \rho_{j-1} * \alpha_j * \rho_j^{-1}$ ($2 \leq j \leq n-1$) y $\beta_n = \rho_{n-1} * \alpha_n$. Cada $\beta_j$ es un lazo basado en $x_0$.

La imagen de $\beta_j$ está en $V_j$: $\alpha_j$ por construcción, $\rho_{j-1}$ en $V_{j-1} \cap V_j \subseteq V_j$ y $\rho_j$ en $V_j \cap V_{j+1} \subseteq V_j$. Así $\beta_j \in C_{x_0}(V_j)$ y, como $V_j$ es simplemente conexo, $\beta_j \simeq_{\{0,1\}} c_{x_0}$ mediante una homotopía en $V_j$; componiéndola con la inclusión $V_j \hookrightarrow X$, también en $X$. Por tanto
\[ [\alpha] = [c_{x_0}] \cdots [c_{x_0}] = [c_{x_0}]. \]
(Si $n = 1$, $\alpha$ ya es un lazo en $V_1$ y se aplica directamente el argumento anterior.)
\end{proof}

\subsection{La esfera $S^n$ es simplemente conexa}

\begin{proposicion}{ [C7.7]}{corolario_sn}
Para $n \ge 2$, $S^n$ es simplemente conexo.
\end{proposicion}

\begin{proof}
Supongamos $n \ge 2$:
Tomamos $V_1 := S^n \setminus \{(0, \dots, 0, 1)\} = S^n \setminus \{N\}$ y $V_2 := S^n \setminus \{(0, \dots, 0, -1)\} = S^n \setminus \{S\}$.
\begin{itemize}
\item $V_1 \cap V_2 = S^n \setminus \{N, S\} \cong S^{n-1} \times \mathopen]-1, 1\mathclose[$. $S^{n-1} \times \mathopen]-1, 1\mathclose[$ son arco-conexos, luego también lo es.
\item $V_1, V_2$ contráctiles $\longrightarrow$ luego son simplemente conexos.
\end{itemize}
Por aplicación directa del teorema anterior concluimos que $S^n$ es simplemente conexo.

\end{proof}

\subsection{La conjetura de Poincaré}

\textbf{Conjetura de Poincaré [C8.1]:}

Sea $X$ un espacio métrico compacto \underline{localmente homeomorfo a $\mathbb{R}^3$} (significa que $\forall p \in X$, existe un entorno abierto de $p$ homeomorfo a algún abierto conexo de $\mathbb{R}^3$; es decir, una 3-variedad).
Supongamos que $X$ es simplemente conexo. Entonces: $X \cong S^3$. Es el recíproco, en dimensión 3, de la \cref{prop:corolario_sn}.

La afirmación análoga en dimensión $n$ ($X$ localmente homeomorfo a $\mathbb{R}^n$ y simplemente conexo implica $X \cong S^n$) fue demostrada, según los casos, por:

\begin{itemize}
    \item Cuando $n\geq 7 \implies$ Stallings (1960)
    \item Cuando $n=5,6 \implies$ Zeeman (1961)
    \item Cuando $n\geq 5 \implies$ Smale (1961, con otra técnica: h-cobordismo)
    \item Cuando $n=4 \implies$ Freedman (1982)
    \item Cuando $n=3 \implies$ G. Perelman (2002--2003, en arXiv)
\end{itemize}

\subsection{Aplicación recubridora y elevaciones}

\textbf{Objetivo [C8.2]:} ver que $\pi_1(S^1) \cong (\mathbb{Z},+)$.

\begin{definicion}{Aplicación exponencial [C8.4]}{}
$p: \mathbb{R} \to S^1$, $p(t) = e^{2\pi i t}$, $\forall t \in \mathbb{R}$ ($S^1 \subseteq \mathbb{R}^2, S^1 \subseteq \mathbb{C}$).
\end{definicion}

\begin{definicion}{Aplicación recubridora (Covering map) [C8.3]}{}
Sea $f: X \to Y$. Diremos que es una aplicación recubridora cuando $\forall y \in Y$ existe un entorno abierto $U$ de $y$ tal que:
\begin{enumerate}
    \item $f^{-1}(U)$ es unión disjunta de abiertos: $f^{-1}(U) = \bigcup_{j \in \Lambda} V_j$.
    \item $f|_{V_j} : V_j \to U$ es un homeomorfismo $\forall j \in \Lambda$.
\end{enumerate}
\end{definicion}

Si $t \in \left]-\frac{1}{2}, \frac{1}{2}\right[$:

\textbf{Lema [C8.6]:}
$p|_{\left]-\frac{1}{2}, \frac{1}{2}\right[} : \left]-\frac{1}{2}, \frac{1}{2}\right[ \to S^1 \setminus \{(-1,0)\}$ es un homeomorfismo.

Denotaremos por $\Psi$ la inversa de esta aplicación, i.e. $\Psi = \left(p|_{\left]-\frac{1}{2}, \frac{1}{2}\right[}\right)^{-1}$.
Es sencillo comprobar que $\Psi(z) = \frac{\arg_0(z)}{2\pi}$.
Recordemos que $\arg_0(z)$ es el único argumento de $z$ que pertenece a $]-\pi, \pi[$ (suponiendo que $z \notin ]-\infty, 0]$).

\begin{definicion}{Elevación o levantamiento [C8.7]}{}
Sea $f: X \to S^1$ continua (con $X$ cualquier espacio topológico). Una elevación (o levantamiento) de $f$ es una aplicación $\tilde{f}: X \to \mathbb{R}$ tal que $p \circ \tilde{f} = f$.
\end{definicion}

\begin{proposicion}{Unicidad de levantamientos [C8.8]}{}
Sea $X$ conexo. Sea $f: X \to S^1$ continua. Sean $f_1, f_2 : X \to \mathbb{R}$ levantamientos de $f$ tales que $f_1(x_0) = f_2(x_0)$ para algún $x_0 \in X$. Entonces $f_1 = f_2$.
\end{proposicion}

\textbf{Demostración:}
$p: \mathbb{R} \to S^1$ es un homomorfismo de grupos:
$p: (\mathbb{R},+) \to (S^1, \cdot)$, ya que $p(t_1+t_2) = e^{2\pi i (t_1+t_2)} = e^{2\pi i t_1} \cdot e^{2\pi i t_2} = p(t_1)p(t_2)$.
$\ker(p) = \{t \in \mathbb{R} \mid e^{2\pi i t} = 1\} = \mathbb{Z}$ (y por tanto $\mathbb{R}/\mathbb{Z} \cong S^1$).

Consideremos $g = f_1 - f_2$.
\[ (p \circ g)(x) = p(f_1(x) - f_2(x)) = e^{2\pi i (f_1(x)-f_2(x))} = \frac{e^{2\pi i f_1(x)}}{e^{2\pi i f_2(x)}} = \frac{p(f_1(x))}{p(f_2(x))} = \frac{f(x)}{f(x)} = 1 \]
Por tanto, $\text{Img}(g) \subseteq p^{-1}(1) = \mathbb{Z}$.

Como $X$ es conexo y $g$ continua $\implies g(X)$ es conexo.
Como $g(X)$ es conexo y $g(X) \subseteq \mathbb{Z} \implies g(x)$ es constante.
Luego $\exists k \in \mathbb{Z}$ tal que $g(x) = k \ \forall x \in X$. Como $g(x_0) = f_1(x_0) - f_2(x_0) = 0$, se tiene que $k=0 \implies f_1(x) = f_2(x) \ \forall x \in X$. \qed

\subsection{Existencia de elevaciones}

\textbf{Teorema existencia de elevaciones:}

\begin{definicion}{Continuidad uniforme [C8.9]}{}
Sea $f: X \subseteq \mathbb{R}^n \to \mathbb{R}^m$. Diremos que $f$ es uniformemente continua cuando $\forall \varepsilon > 0$, $\exists \delta > 0$ tal que $\forall x, x' \in X$ que cumplan $\|x-x'\| < \delta$, se verifica que $\|f(x)-f(x')\| < \varepsilon$.
\end{definicion}

Es obvio que $f$ uniformemente continua $\implies f$ continua (en todos los puntos de $X$). Esto es falso en general, pero es cierto si $X$ es compacto.

\textbf{Lema [C8.10]:}
Sea $f: X \subseteq \mathbb{R}^n \to \mathbb{R}^m$ continua, con $X$ compacto $\implies f$ uniformemente continua.

\textbf{Demostración:}
Sea $\varepsilon > 0$. Sea $x \in X$. Por hipótesis:
$\exists \delta_x > 0$ tal que $\forall x' \in X$ con $\|x-x'\| < \delta_x$, se cumple que $\|f(x)-f(x')\| < \varepsilon/2$. ($B(x,\delta_x) = \{y \in X \mid \|x-y\| < \delta_x\}$, bola abierta de centro $x$, radio $\delta_x$).
Claramente, $X = \bigcup_{x\in X} B(x, \delta_x/2)$.
Como $X$ es compacto, admite un subrecubrimiento finito: $\exists x_1, \dots, x_r \in X$ tales que $X = \bigcup_{j=1}^r B(x_j, \delta_{x_j}/2)$.
Sea $\delta = \min \left\{\frac{\delta_{x_1}}{2}, \dots, \frac{\delta_{x_r}}{2}\right\}$.
Veamos que $\delta$ funciona:
Sea $x,x' \in X$ con $\|x-x'\| < \delta$. Veamos que $\|f(x)-f(x')\| < \varepsilon$.
Dado que $x \in X$, $\exists j \in \{1,\dots,r\}$ tal que $x \in B(x_j, \delta_{x_j}/2)$, por tanto, $\|x-x_j\| < \delta_{x_j}/2$.
Observamos que:
\[ \|x'-x_j\| \leq \|x'-x\| + \|x-x_j\| < \delta + \frac{\delta_{x_j}}{2} \leq \frac{\delta_{x_j}}{2} + \frac{\delta_{x_j}}{2} = \delta_{x_j} \implies \|f(x')-f(x_j)\| < \frac{\varepsilon}{2} \]
Finalmente:
\[ \|f(x)-f(x')\| \leq \|f(x)-f(x_j)\| + \|f(x_j)-f(x')\| < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon \] \qed

\begin{teorema}{Existencia de elevaciones [C8.11]}{teo_existencia_elevaciones}
Sea $X \subseteq \mathbb{R}^n$. Supongamos que $X$ es convexo y compacto. Sea $f: X \to S^1$ continua. Sea $x_0 \in X$. Sea $t_0 \in \mathbb{R}$ tal que $p(t_0) = f(x_0)$. Entonces existe una única elevación $\tilde{f} : X \to \mathbb{R}$ de $f$ tal que $\tilde{f}(x_0) = t_0$.
\end{teorema}

\textbf{Demostración:}
En virtud del teorema anterior se deduce automáticamente la unicidad (ya que $X$ es conexo).
Veamos la existencia de $\tilde{f}$:
\begin{enumerate}
    \item $f$ es uniformemente continua (ya que $X$ es compacto). Como consecuencia: $\exists \delta > 0$ tal que $\forall x,x' \in X$ si $\|x-x'\| < \delta \implies \|f(x)-f(x')\| < 2$.
    \item $X$ está acotado (por ser compacto): $\exists k > 0$ tal que $\|x\| < k, \forall x \in X \implies \frac{\|x\|}{\delta} < \frac{k}{\delta}, \forall x \in X$. Sea $m \in \mathbb{Z}_{\geq 1}$ tal que $\frac{k}{\delta} < m$. Por consiguiente $\frac{\|x\|}{\delta} < \frac{k}{\delta} < m, \forall x \in X$, para algún $m \in \mathbb{Z}_{\geq 1}$.
    \item Sea $j \in \{0,1,\dots,m-1\}$. Definimos $g_j: X \to S^1 \setminus \{(-1,0)\}$ como:
    \[ g_j(x) = \frac{f\left(\frac{j+1}{m}x\right)}{f\left(\frac{j}{m}x\right)}, \quad \forall x \in X \]
    ¿Puede valer $0$ la función $f(\frac{j}{m}x)$? $\implies$ No ($S^1 \not\ni 0$).
    ¿Puede valer $-1$? $\implies$ Sí, si son antipodales, pero su diferencia es $<2$ luego no lo son.
    \item Sea $j \in \{0,\dots,m-1\}$. Sea $x$ cualquier punto de $X$. Observamos que:
    \[ \left\|\frac{j+1}{m}x - \frac{j}{m}x\right\| = \frac{\|x\|}{m} < \delta \implies \left\|f\left(\frac{j+1}{m}x\right) - f\left(\frac{j}{m}x\right)\right\| < 2 \]
    Sabemos que $\Psi = \left(p|_{\left]-\frac{1}{2}, \frac{1}{2}\right[}\right)^{-1} : S^1 \setminus \{(-1,0)\} \to \left]-\frac{1}{2}, \frac{1}{2}\right[$.
    \item Defino $\tilde{f}: X \to \mathbb{R}$ como:
    \[ \tilde{f}(x) = t_0 + \Psi(g_0(x)) + \Psi(g_1(x)) + \dots + \Psi(g_{m-1}(x)) = t_0 + \sum_{j=0}^{m-1} \Psi(g_j(x)) \]
    \item $\tilde{f}$ cumple 1 y 2:
    \begin{itemize}
        \item (1) $\tilde{f}(x_0) = t_0 + \sum_{j=0}^{m-1} \Psi(g_j(x_0)) = t_0 + \sum_{j=0}^{m-1} \Psi(1) = t_0$. (A partir de ahora suponemos $x_0 = 0$. Veremos luego el caso general (*)).
        \item (2) 
        \begin{align*}
        (p \circ \tilde{f})(x) &= p(t_0) \cdot p(\Psi(g_0(x))) \cdot p(\Psi(g_1(x))) \dots p(\Psi(g_{m-1}(x))) \\
        &= f(0) \cdot g_0(x) \cdot g_1(x) \dots g_{m-1}(x) \\
        &= f(0) \cdot \frac{f(\frac{1}{m}x)}{f(0)} \cdot \frac{f(\frac{2}{m}x)}{f(\frac{1}{m}x)} \dots \frac{f(\frac{m}{m}x)}{f(\frac{m-1}{m}x)} = f(x)
        \end{align*}
    \end{itemize}
\end{enumerate}

(*) Veamos el caso general. Veamos que, suponiendo que hemos encontrado $\tilde{g}: X \to \mathbb{R}$ que cumple las condiciones requeridas para $x_0 = 0$, automáticamente se cumple la existencia de una aplicación verificando 1 y 2 partiendo de un $x_0$ cualquiera no necesariamente igual a 0.

Por tanto, supongamos que $x_0$ es ahora cualquier punto de $X$.
Sea $L: \mathbb{R}^n \to \mathbb{R}^n$ dada por $x \mapsto x - x_0$ (traslación).
Como $X$ es convexo y compacto $\implies L(X)$ también es convexo y compacto.
Además $x_0 \in X \implies L(x_0) = 0 \in L(X)$.

Sea $g = f \circ L^{-1} : L(X) \to S^1$.
De acuerdo con la prueba anterior: $\exists \tilde{g}: L(X) \to \mathbb{R}$ que cumple $\tilde{g}(0) = t_0$ y $p \circ \tilde{g} = g$.

Definimos $\tilde{f} = \tilde{g} \circ L$. Solo queda comprobar que $\tilde{f}$ cumple:
\begin{itemize}
    \item (2) $p \circ \tilde{f} = p \circ (\tilde{g} \circ L) = (p \circ \tilde{g}) \circ L = g \circ L = (f \circ L^{-1}) \circ L = f$.
    \item (1) $\tilde{f}(x_0) = (\tilde{g} \circ L)(x_0) = \tilde{g}(L(x_0)) = \tilde{g}(0) = t_0$. \qed
\end{itemize}

\subsection{El grupo fundamental de $S^1$}

El objetivo es probar que $\pi_1(S^1) \cong (\mathbb{Z},+)$ (\cref{thm:teo_pi1_s1}).

\begin{proposicion}{Elevación de arcos [C9.1]}{cor_elev_arcos}
Sea $\alpha \colon I \to S^1$ continua tal que $\alpha(0) = (1,0)$. Entonces existe una única elevación $\tilde{\alpha} \colon I \to \mathbb{R}$ de $\alpha$ tal que $\tilde{\alpha}(0) = 0$.
\end{proposicion}

\begin{proposicion}{Elevación de homotopías [C9.2]}{cor_elev_homotopias}
Sea $F \colon I \times I \to S^1$ continua tal que $F(0,0) = (1,0)$. Entonces existe una única elevación $\tilde{F} \colon I \times I \to \mathbb{R}$ de $F$ tal que $\tilde{F}(0,0) = 0$.
\end{proposicion}

Ambos resultados se deducen del \cref{thm:teo_existencia_elevaciones}, ya que $I$ e $I \times I$ son convexos y compactos.

\begin{definicion}{Grado de un lazo [C9.3]}{def_grado}
Sea $\alpha \in C_{(1,0)}(S^1)$. Se define el \textbf{grado} de $\alpha$ como $\deg(\alpha) := \tilde{\alpha}(1)$, siendo $\tilde{\alpha}$ la única elevación de $\alpha$ con $\tilde{\alpha}(0) = 0$. Está bien definido por la elevación de arcos.
\end{definicion}

Geométricamente, $\deg(\alpha)$ es el número de vueltas completas de $\alpha$ en sentido antihorario. Como $p(\tilde{\alpha}(1)) = \alpha(1) = (1,0)$, se tiene $\tilde{\alpha}(1) \in p^{-1}(1,0) = \mathbb{Z}$, luego $\deg(\alpha) \in \mathbb{Z}$.

\textbf{Ejemplo.} Para $n \in \mathbb{Z}$, sea $\omega_n \colon I \to S^1$, $\omega_n(t) = e^{2\pi i n t}$. Su elevación con $\tilde{\omega}_n(0)=0$ es $\tilde{\omega}_n(t) = nt$, luego $\deg(\omega_n) = n$.

\begin{teorema}{$\pi_1(S^1)\cong \mathbb{Z}$ [C9.5]}{teo_pi1_s1}
La aplicación $\deg \colon \pi_1(S^1,(1,0)) \to \mathbb{Z}$, $[\alpha] \mapsto \deg(\alpha)$, es un isomorfismo de grupos.
\end{teorema}

\begin{proof}
\begin{enumerate}
    \item \textbf{Está bien definida.} Sean $\alpha, \beta \in C_{(1,0)}(S^1)$ con $\alpha \simeq_{\{0,1\}} \beta$ mediante $F \colon I \times I \to S^1$:
    \[ F(s,0) = \alpha(s), \quad F(s,1) = \beta(s), \quad F(0,t) = F(1,t) = (1,0) \quad \forall s,t \in I. \]
    Sea $\tilde{F}$ la elevación de $F$ con $\tilde{F}(0,0) = 0$. Entonces $s \mapsto \tilde{F}(s,0)$ es la elevación de $\alpha$ que vale $0$ en $0$, es decir $\tilde{\alpha}$, y $s \mapsto \tilde{F}(s,1)$ es la elevación de $\beta$ que vale $0$ en $0$ (pues $\tilde{F}(0,t)\in p^{-1}(1,0)=\mathbb{Z}$ es constante por conexión y vale $\tilde{F}(0,0)=0$). Por otro lado, $\tilde{F}(\{1\} \times I) \subseteq p^{-1}(1,0) = \mathbb{Z}$ es conexo, luego $\tilde{F}(1,t) = k$ para cierto $k \in \mathbb{Z}$ y todo $t$. Así
    \[ \deg(\alpha) = \tilde{\alpha}(1) = \tilde{F}(1,0) = k = \tilde{F}(1,1) = \tilde{\beta}(1) = \deg(\beta). \]

    \item \textbf{Es homomorfismo.} Sean $\alpha, \beta \in C_{(1,0)}(S^1)$ con elevaciones $\tilde{\alpha}, \tilde{\beta}$ que valen $0$ en $0$. Definimos $w \colon I \to \mathbb{R}$ por
    \[ w(s) = \begin{cases} \tilde{\alpha}(2s), & s \in [0,\tfrac12], \\ \tilde{\alpha}(1) + \tilde{\beta}(2s-1), & s \in [\tfrac12,1]. \end{cases} \]
    Es continua (en $s=\tfrac12$ ambas valen $\tilde{\alpha}(1)$), $w(0)=0$ y $p \circ w = \alpha * \beta$, pues $p(\tilde{\alpha}(1) + \tilde{\beta}(u)) = p(\tilde{\alpha}(1))\,p(\tilde{\beta}(u)) = \beta(u)$ ya que $p(\tilde{\alpha}(1)) = (1,0)$. Luego $w$ es la elevación de $\alpha*\beta$ que empieza en $0$ y
    \[ \deg(\alpha * \beta) = w(1) = \tilde{\alpha}(1) + \tilde{\beta}(1) = \deg(\alpha) + \deg(\beta). \]

    \item \textbf{Inyectiva.} Sean $\alpha, \beta \in C_{(1,0)}(S^1)$ con $\deg(\alpha) = \deg(\beta)$. Consideramos la homotopía convexa entre sus elevaciones
    \[ H \colon I \times I \to \mathbb{R}, \quad H(s,t) = (1-t)\tilde{\alpha}(s) + t\tilde{\beta}(s). \]
    Entonces $H(0,t) = 0$ y $H(1,t) = (1-t)\deg(\alpha) + t\deg(\beta) = \deg(\alpha)$ no dependen de $t$, es decir, $H \colon \tilde{\alpha} \simeq_{\{0,1\}} \tilde{\beta}$. Sea $F = p \circ H$. Entonces $F_0 = p\circ\tilde{\alpha} = \alpha$, $F_1 = \beta$, $F(0,t) = p(0) = (1,0)$ y $F(1,t) = p(\deg\alpha) = (1,0)$, luego $F \colon \alpha \simeq_{\{0,1\}} \beta$.

    \item \textbf{Sobreyectiva.} Dado $n \in \mathbb{Z}$, el lazo $\omega_n$ cumple $\deg([\omega_n]) = n$ (ejemplo anterior).
\end{enumerate}
\end{proof}

\begin{proposicion}{$S^1$ no es contráctil [C9.6]}{cor_s1_no_contractil}
$S^1$ no es contráctil, ya que $\pi_1(S^1) \cong \mathbb{Z} \neq 1$.
\end{proposicion}

\begin{proof}
Supongamos que $S^1$ es contráctil. Por la caracterización de contráctiles (\cref{thm:teo_contractil}), $S^1$ tiene el mismo tipo de homotopía que un espacio $\{p\}$ formado por un único punto. Como $S^1$ y $\{p\}$ son arco-conexos y el grupo fundamental es un invariante homotópico (\cref{prop:equiv_homotopica}), se tiene $\pi_1(S^1) \cong \pi_1(\{p\}) = 1$, es decir, $\mathbb{Z} \cong 1$, lo cual es absurdo. Luego $S^1$ no es contráctil.
\end{proof}

\begin{proposicion}{$S^2$ y $T^2$ no son homeomorfos [C9.7]}{cor_s2_t2}
$S^2$ y $T^2$ no son homeomorfos.
\end{proposicion}

\begin{proof}
$S^2$ es simplemente conexo, así que $\pi_1(S^2) = 1$, mientras que $\pi_1(T^2) \cong \mathbb{Z} \times \mathbb{Z}$ (véase el ejercicio siguiente con $T^2 = S^1 \times S^1$). Como el grupo fundamental es un invariante topológico, $S^2 \not\cong T^2$.
\end{proof}

\textbf{Ejercicio [C9.8].} Sean $X$ e $Y$ espacios topológicos y $(x_0,y_0) \in X \times Y$. Entonces
\[ \pi_1(X \times Y, (x_0,y_0)) \cong \pi_1(X,x_0) \times \pi_1(Y,y_0). \]
En particular, si $X$ e $Y$ son arco-conexos, $\pi_1(X \times Y) \cong \pi_1(X) \times \pi_1(Y)$.

\begin{observacion}{Grupo fundamental del cilindro [C9.9]}{obs_cilindro}
El grupo fundamental del cilindro $S^1 \times I$ es $\mathbb{Z}$, ya que $S^1$ y $S^1 \times I$ son del mismo tipo de homotopía.
\end{observacion}

\begin{ejercicio}{Tema 4, ej. 1 (esferas tangentes)}
Sean $S, S' \subseteq \mathbb{R}^3$ dos esferas del mismo radio que se cortan en un único punto $x_0$ (son tangentes). Probar que $A = S \cup S'$ es simplemente conexo, y lo mismo para una cadena finita de esferas $S_1, \dots, S_r$ con $S_i$ y $S_{i+1}$ tangentes y $S_1 \cap S_r = \emptyset$.

\textbf{Solución.} Se aplica la \cref{thm:cond_simp_conexo} con un recubrimiento abierto de $A$ por dos conjuntos. Hay que tener cuidado con la topología: una esfera sola \emph{no} es abierta en $A$. Recordemos que $Y \subseteq A$ es abierto en $A$ (topología relativa) cuando para todo $p \in Y$ existe $\varepsilon > 0$ con $B(p,\varepsilon) \cap A \subseteq Y$.

Sean $q \in S \setminus \{x_0\}$ y $q' \in S' \setminus \{x_0\}$. Tomamos
\[ V_1 = S \cup (S' \setminus \{q'\}), \qquad V_2 = (S \setminus \{q\}) \cup S'. \]
Ambos son abiertos en $A$ (su complementario es un punto) y $V_1 \cup V_2 = A$.
\begin{itemize}
    \item $V_1$ es una esfera con un disco abierto (contráctil, pues $S' \setminus \{q'\} \cong \mathbb{R}^2$) pegado por un punto, luego $V_1 \simeq S^2$, que es simplemente conexo. Igual con $V_2$.
    \item $V_1 \cap V_2 = (S \setminus \{q\}) \cup (S' \setminus \{q'\})$ son dos copias de $\mathbb{R}^2$ unidas por $x_0$, que es arco-conexo.
    \item $x_0 \in V_1 \cap V_2$.
\end{itemize}
Por el teorema, $A$ es simplemente conexo.

Para la cadena de $r$ esferas se razona por \textbf{inducción} sobre $r$, añadiendo una esfera tangente en cada paso. La hipótesis $S_1 \cap S_r = \emptyset$ es necesaria: si la cadena se cierra formando un anillo de esferas, el espacio \textbf{no} es simplemente conexo.
\end{ejercicio}

\begin{ejercicio}{Tema 4, ej. 2, apartados a)--d) (grupo fundamental de subespacios)}
En cada apartado se busca un espacio $Y$ del mismo tipo de homotopía que $X$ cuyo grupo fundamental sea conocido o fácil de deducir. Aunque el enunciado no lo pide, conviene explicitar una equivalencia homotópica entre $X$ e $Y$.
\begin{enumerate}
    \item[a)] $X = \{(x,y) \in \mathbb{R}^2 : 1 \leq x^2 + y^2 \leq 4\}$ (corona). Se tiene $X \simeq S^1$ y, por la invariancia homotópica, $\pi_1(X) \cong \pi_1(S^1) \cong \mathbb{Z}$. La equivalencia homotópica es
    \[ f \colon S^1 \to X, \ (x,y) \mapsto (x,y), \qquad g \colon X \to S^1, \ (x,y) \mapsto \frac{(x,y)}{\|(x,y)\|}. \]
    \item[b)] $X = \mathbb{R}^2 \setminus \{(0,0)\}$. $\pi_1(X) \cong \pi_1(S^1) \cong \mathbb{Z}$.
    \item[c)] $X = \{(x,f(x)) : x \in [a,b]\}$, con $f$ continua. Es \textbf{contráctil} (homeomorfo a $[a,b]$), luego simplemente conexo.
    \item[d)] $X = S^1 \cup \{y^2 = 1\}$ (la circunferencia y las dos rectas horizontales $y = \pm 1$, tangentes a ella). $\pi_1(X) \cong \pi_1(S^1) \cong \mathbb{Z}$.
\end{enumerate}
\end{ejercicio}
