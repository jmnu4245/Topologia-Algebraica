\section{Espacios simplemente conexos, deformación de aplicaciones y grupo fundamental de $S^n$}

\subsection{Espacios simplemente conexos}

\begin{observacion}{}{}
Por la \cref{prop:equiv_homotopica}, si $X$ es contráctil (es decir, $X \simeq \{p\}$), entonces $\pi_1(X) = 1$.
\end{observacion}

\begin{definicion}{Espacio simplemente conexo [C7.1]}{def_simp_conexo}
Sea $X$ un espacio topológico. Diremos que $X$ es simplemente conexo cuando es arco-conexo y $\pi_1(X) = 1$.
\end{definicion}

\begin{proposicion}{ [C7.2] \marcaP}{prop_arco_conexo}
Sea $X$ un espacio arco-conexo. Entonces:

$$\pi_1(X) = 1 \iff \forall x_0, x_1 \in X, \forall \alpha, \beta \in C_{x_0, x_1}(X) : \alpha \simeq_{\{0,1\}} \beta$$
\end{proposicion}

A la izquierda, $\alpha$ se deforma en $\beta$ (líneas discontinuas) sin mover los extremos. A la derecha, el espacio también es arco-conexo, pero el agujero impide la deformación: los caminos intermedios bajan hasta tocar el agujero ($\times$) y no pueden pasar al otro lado sin atravesarlo. Por eso la arco-conexión no basta y hace falta $\pi_1 = 1$: el lazo $\alpha * \beta^{-1}$ rodea el agujero y no es trivial.


\begin{proof}
$\Rightarrow)$ Sean $x_0, x_1 \in X$ y $\alpha, \beta \in C_{x_0, x_1}(X)$. El lazo $\alpha * \beta^{-1}$ está en $C_{x_0}(X)$ y, como $\pi_1(X, x_0) \cong \pi_1(X) = 1$, se tiene $[\alpha * \beta^{-1}] = [c_{x_0}]$. Usando el producto de clases de arcos (\cref{obs:obs_producto_arcos}):
\[ [\alpha] = [\alpha] \cdot [\beta^{-1}] \cdot [\beta] = [\alpha * \beta^{-1}] \cdot [\beta] = [c_{x_0}] \cdot [\beta] = [\beta], \]
es decir, $\alpha \simeq_{\{0,1\}} \beta$.

$\Leftarrow)$ Sea $\alpha \in C_{x_0}(X)$. Aplicando la hipótesis con $x_1 = x_0$ y $\beta = c_{x_0}$, se obtiene $\alpha \simeq_{\{0,1\}} c_{x_0}$, luego $\pi_1(X,x_0) = 1$.
\end{proof}

\subsection{Reparametrización de arcos y lema de Lebesgue}

\begin{proposicion}{Reparametrización [C7.3] \marcaE}{prop_reparametrizacion}
Sea $g: I \longrightarrow I$ una aplicación continua tal que $g(0)=0$ y $g(1)=1$. Sea $\alpha: I \longrightarrow X$ una aplicación continua, siendo $X$ un espacio topológico cualquiera. Entonces se cumple que $\alpha \simeq_{\{0,1\}} \alpha \circ g$.
\end{proposicion}
\begin{center}
\begin{tikzpicture}[scale=1.5]
  \draw[->] (-0.2,0) -- (1.5,0);
  \draw[->] (0,-0.2) -- (0,1.5);
  \draw (1,0.05) -- (1,-0.05) node[below] {$1$};
  \draw (0.05,1) -- (-0.05,1) node[left] {$1$};
  \draw[blue, thick] (0,0) .. controls (0.2,0.8) and (0.6,0.3) .. (1,1);
\end{tikzpicture}
\captionof{figure}{Gráfica de la reparametrización continua $g: I \longrightarrow I$.}
\end{center}
\begin{proof}
Sea $F: I \times I \longrightarrow X$ la aplicación continua definida por $F(s,t) = \alpha((1-t)s + t g(s))$. Comprobamos sus valores:
\begin{itemize}
\item $F_0(s) = F(s,0) = \alpha((1-0)s + 0 \cdot g(s)) = \alpha(s)$, $\forall s \in I$.
\item $F_1(s) = F(s,1) = \alpha((1-1)s + 1 \cdot g(s)) = \alpha(g(s))$, $\forall s \in I$.
\item $F(0,t) = \alpha((1-t)0 + t g(0)) = \alpha(0)$, que no depende de $t$.
\item $F(1,t) = \alpha((1-t)1 + t g(1)) = \alpha((1-t) + t) = \alpha(1)$, que tampoco depende de $t$.
\end{itemize}
\end{proof}

\begin{teorema}{Lema de Lebesgue [C7.4]}{lema_lebesgue}
Sea $(X,d)$ un espacio métrico compacto y sea $\mathcal{U} = \{U_i\}_{i \in \Lambda}$ un recubrimiento abierto de $X$ (lo que significa que $U_i$ es abierto en $X$ y además $\bigcup_{i \in \Lambda} U_i = X$). Entonces, $\exists \delta > 0$ tal que $\forall x \in X, \exists i(x) \in \Lambda$ verificando que $B(x,\delta) \subseteq U_{i(x)}$.
\end{teorema}

\begin{proof}
Dado que $\bigcup_{i \in \Lambda} U_i = X$, se tiene que $\forall x \in X, \exists r(x) > 0$ tal que $B(x, r(x)) \subseteq U_{i(x)}$ para algún $i(x) \in \Lambda$.
Definimos la familia $\mathcal{F} = \{B(x, \frac{r(x)}{2})\}_{x \in X}$. Claramente $\mathcal{F}$ es un recubrimiento abierto de $X$. Como $X$ es compacto, admite un subrecubrimiento finito de la forma:

$$X = B\left(x_1, \frac{r(x_1)}{2}\right) \cup \dots \cup B\left(x_n, \frac{r(x_n)}{2}\right).$$
Sea $\delta = \min \left\{ \frac{r(x_1)}{2}, \dots, \frac{r(x_n)}{2} \right\}$.
Sea $x \in X$. Existe un índice $j \in \{1, \dots, n\}$ tal que $x \in B(x_j, \frac{r(x_j)}{2})$. Veamos que $B(x,\delta) \subseteq U_{i(x_j)}$:
Si $y \in B(x,\delta)$, entonces

$$d(y, x_j) \le d(y,x) + d(x,x_j) < \delta + \frac{r(x_j)}{2} \le \frac{r(x_j)}{2} + \frac{r(x_j)}{2} = r(x_j).$$
Hemos llegado a la conclusión de que $B(x,\delta) \subseteq B(x_j, r(x_j)) \subseteq U_{i(x_j)}$.
\end{proof}

\begin{proposicion}{Lebesgue para recubrimientos de $[0,1]$ [C7.5] \marcaC}{cor_lebesgue_I}
Sea $\mathcal{U}$ un recubrimiento abierto de $I = [0,1]$. Existe $n \geq 1$ tal que cada $\left[\frac{j-1}{n}, \frac{j}{n}\right]$ ($j = 1,\dots,n$) está contenido en algún elemento de $\mathcal{U}$: la partición subordinada a $\mathcal{U}$ puede tomarse \textbf{regular}.
\end{proposicion}

\begin{proof}
Sea $\delta > 0$ el dado por el lema de Lebesgue (\cref{thm:lema_lebesgue}) y sea $n$ con $\frac1n < \delta$. Si $x_j$ es el punto medio de $[t_{j-1},t_j]$, con $t_j = \frac jn$, entonces $[t_{j-1},t_j] \subseteq \left]x_j - \delta, x_j + \delta\right[ \cap I$, que está contenido en algún $U \in \mathcal{U}$.
\end{proof}

\subsection{Condición suficiente para ser simplemente conexo}

\begin{teorema}{Condición suficiente para que un espacio sea simplemente conexo [C7.6] \marcaT}{cond_simp_conexo}
Sea $X$ un espacio topológico y sea $\{V_i\}_{i \in \Lambda}$ un recubrimiento abierto de $X$. Supongamos que:
\begin{itemize}
\item[a)] $V_i$ es simplemente conexo para todo $i \in \Lambda$.
\item[b)] Para todo $i \neq j$, se cumple que $V_i \cap V_j$ es arco-conexo.
\item[c)] $\bigcap_{i \in \Lambda} V_i \neq \emptyset$.
\end{itemize}
Entonces, $X$ es simplemente conexo. 
\end{teorema}


\begin{proof}
$X$ es arco-conexo: los $V_i$ lo son (por a) y tienen un punto común (por c). Sea $x_0 \in \bigcap_i V_i$ y sea $\alpha \in C_{x_0}(X)$; veamos que $\alpha \simeq_{\{0,1\}} c_{x_0}$.

Los conjuntos $\alpha^{-1}(V_i)$ son abiertos que recubren $I$, así que, por la \cref{prop:cor_lebesgue_I}, existe $n$ tal que, con $t_j = \frac jn$, $\alpha([t_{j-1},t_j]) \subseteq V_j$ para cierto $V_j$ del recubrimiento. En particular, $x_0, \alpha(t_j) \in V_j \cap V_{j+1}$ para $j = 1,\dots,n-1$.

Llamamos $\alpha_j \colon I \to V_j$, $\alpha_j(s) = \alpha(t_{j-1} + \frac sn)$, al tramo $j$-ésimo de $\alpha$ reparametrizado a $I$. Como la partición es regular, $\alpha = \alpha_1 * \dots * \alpha_n$: en el tramo $j$ la concatenación conjunta (\cref{def:concat_conjunta}) vale $\alpha_j(nu - (j-1)) = \alpha(u)$.

Para cerrar cada tramo en $x_0$ usamos que $V_j \cap V_{j+1}$ es arco-conexo (por b; si $V_j = V_{j+1}$, por a): existe $\rho_j \colon I \to V_j \cap V_{j+1}$ de $x_0$ a $\alpha(t_j)$. Ponemos además $\rho_0 = \rho_n = c_{x_0}$.

\[ [\alpha] = [\alpha_1 * \rho_1^{-1} * \rho_1 * \alpha_2 * \dots * \rho_{n-1}^{-1} * \rho_{n-1} * \alpha_n] = [\beta_1] \cdots [\beta_n], \qquad \beta_j = \rho_{j-1} * \alpha_j * \rho_j^{-1}. \]
Pero cada $\beta_j$ es un lazo en $x_0$ contenido en $V_j$, pues $\rho_{j-1}$, $\alpha_j$ y $\rho_j$ lo están, y como $V_j$ es simplemente conexo, $\beta_j \simeq_{\{0,1\}} c_{x_0}$ en $V_j$ y, por tanto, en $X$. Así $[\alpha] = [c_{x_0}]$.
\end{proof}

\subsection{La esfera $S^n$ es simplemente conexa}

\begin{proposicion}{ [C7.7] \marcaC}{corolario_sn}
Para $n \ge 2$, $S^n$ es simplemente conexo.
\end{proposicion}

\begin{proof}
Supongamos $n \ge 2$:
Tomamos $V_1 := S^n \setminus \{(0, \dots, 0, 1)\} = S^n \setminus \{N\}$ y $V_2 := S^n \setminus \{(0, \dots, 0, -1)\} = S^n \setminus \{S\}$.
\begin{itemize}
\item $V_1 \cap V_2 = S^n \setminus \{N, S\} \cong S^{n-1} \times \mathopen]-1, 1\mathclose[$. $S^{n-1} \times \mathopen]-1, 1\mathclose[$ son arco-conexos, luego también lo es.
\item $V_1, V_2$ contráctiles $\longrightarrow$ luego son simplemente conexos.
\end{itemize}
Por aplicación directa del teorema anterior concluimos que $S^n$ es simplemente conexo.

\end{proof}

\subsection{La conjetura de Poincaré}

\textbf{Conjetura de Poincaré [C8.1]:}

Sea $X$ un espacio métrico compacto \underline{localmente homeomorfo a $\mathbb{R}^3$} (significa que $\forall p \in X$, existe un entorno abierto de $p$ homeomorfo a algún abierto conexo de $\mathbb{R}^3$; es decir, una 3-variedad).
Supongamos que $X$ es simplemente conexo. Entonces: $X \cong S^3$. Es el recíproco, en dimensión 3, de la \cref{prop:corolario_sn}.

La afirmación análoga en dimensión $n$ ($X$ localmente homeomorfo a $\mathbb{R}^n$ y simplemente conexo implica $X \cong S^n$) fue demostrada, según los casos, por:

\begin{itemize}
    \item Cuando $n\geq 7 \implies$ Stallings (1960)
    \item Cuando $n=5,6 \implies$ Zeeman (1961)
    \item Cuando $n\geq 5 \implies$ Smale (1961, con otra técnica: h-cobordismo)
    \item Cuando $n=4 \implies$ Freedman (1982)
    \item Cuando $n=3 \implies$ G. Perelman (2002--2003, en arXiv)
\end{itemize}

\subsection{Aplicación recubridora y elevaciones}

\textbf{Objetivo [C8.2]:} ver que $\pi_1(S^1) \cong (\mathbb{Z},+)$.

\begin{definicion}{Aplicación recubridora (Covering map) [C8.3]}{def_recubridora}
Sea $f: X \to Y$. Diremos que es una aplicación recubridora cuando $\forall y \in Y$ existe un entorno abierto $U$ de $y$ tal que:
\begin{enumerate}
    \item $f^{-1}(U)$ es unión disjunta de abiertos: $f^{-1}(U) = \bigcup_{j \in \Lambda} V_j$.
    \item $f|_{V_j} : V_j \to U$ es un homeomorfismo $\forall j \in \Lambda$.
\end{enumerate}
\end{definicion}

\begin{definicion}{Aplicación exponencial [C8.4]}{def_exponencial}
$p: \mathbb{R} \to S^1$, $p(t) = e^{2\pi i t}$, $\forall t \in \mathbb{R}$ ($S^1 \subseteq \mathbb{R}^2, S^1 \subseteq \mathbb{C}$).

Se cumple que $p$ es una aplicación recubridora. Para $U = S^1 \setminus \{(-1,0)\}$,
\[ p^{-1}(U) = \bigsqcup_{k \in \mathbb{Z}} \left] k - \tfrac12,\, k + \tfrac12 \right[, \]
y $p$ aplica cada uno de esos intervalos homeomórficamente sobre $U$ (por el lema siguiente y las traslaciones $t \mapsto t + k$). Para $y = (-1,0)$ se toma $U = S^1 \setminus \{(1,0)\}$, con los intervalos $]k, k+1[$.
\end{definicion}

\begin{observacion}{$p$ es homomorfismo [C8.5]}{obs_p_homomorfismo}
$p$ es un homomorfismo sobreyectivo de $(\mathbb{R},+)$ en $(S^1,\cdot)$, pues $p(t_1+t_2) = e^{2\pi i t_1} e^{2\pi i t_2} = p(t_1)p(t_2)$, y $p^{-1}(1) = \ker p = \mathbb{Z}$. En particular, $\mathbb{R}/\mathbb{Z} \cong S^1$ como grupos. Todas las fibras $p^{-1}(z) = t + \mathbb{Z}$ son discretas: los ``pisos'' de la hélice están separados.
\end{observacion}

\begin{proposicion}{Lema (restricción de $p$) [C8.6]}{lema_restriccion_p}
$p|_{\left]-\frac{1}{2}, \frac{1}{2}\right[} : \left]-\frac{1}{2}, \frac{1}{2}\right[ \to S^1 \setminus \{(-1,0)\}$ es un homeomorfismo.
\end{proposicion}

Denotaremos por $\Psi$ la inversa de esta aplicación, i.e. $\Psi = \left(p|_{\left]-\frac{1}{2}, \frac{1}{2}\right[}\right)^{-1}$.
Es sencillo comprobar que $\Psi(z) = \frac{\arg_0(z)}{2\pi}$.
Recordemos que $\arg_0(z)$ es el único argumento de $z$ que pertenece a $]-\pi, \pi[$ (suponiendo que $z \notin ]-\infty, 0]$).

\begin{definicion}{Elevación o levantamiento [C8.7]}{def_elevacion}
Sea $f: X \to S^1$ continua (con $X$ cualquier espacio topológico). Una elevación (o levantamiento) de $f$ es una aplicación \textbf{continua} $\tilde{f}: X \to \mathbb{R}$ tal que $p \circ \tilde{f} = f$.
\end{definicion}

\begin{proposicion}{Unicidad de levantamientos [C8.8] \marcaT}{prop_unicidad_elev}
Sea $X$ conexo. Sea $f: X \to S^1$ continua. Sean $f_1, f_2 : X \to \mathbb{R}$ levantamientos de $f$ tales que $f_1(x_0) = f_2(x_0)$ para algún $x_0 \in X$. Entonces $f_1 = f_2$.
\end{proposicion}

\begin{proof}
Usamos que $p$ es un homomorfismo de $(\mathbb{R},+)$ en $(S^1,\cdot)$ con $p^{-1}(1) = \ker p = \mathbb{Z}$ (\cref{obs:obs_p_homomorfismo}). Consideremos $g = f_1 - f_2$, que es continua.
\[ (p \circ g)(x) = p(f_1(x) - f_2(x)) = e^{2\pi i (f_1(x)-f_2(x))} = \frac{e^{2\pi i f_1(x)}}{e^{2\pi i f_2(x)}} = \frac{p(f_1(x))}{p(f_2(x))} = \frac{f(x)}{f(x)} = 1 \]
Por tanto, $\text{Img}(g) \subseteq p^{-1}(1) = \mathbb{Z}$.

Como $X$ es conexo y $g$ continua $\implies g(X)$ es conexo.
Como $g(X)$ es conexo y $g(X) \subseteq \mathbb{Z} \implies g(x)$ es constante.
Luego $\exists k \in \mathbb{Z}$ tal que $g(x) = k \ \forall x \in X$. Como $g(x_0) = f_1(x_0) - f_2(x_0) = 0$, se tiene que $k=0 \implies f_1(x) = f_2(x) \ \forall x \in X$.
\end{proof}

\subsection{Existencia de elevaciones}

\begin{definicion}{Continuidad uniforme [C8.9]}{}
Sea $f: X \subseteq \mathbb{R}^n \to \mathbb{R}^m$. Diremos que $f$ es uniformemente continua cuando $\forall \varepsilon > 0$, $\exists \delta > 0$ tal que $\forall x, x' \in X$ que cumplan $\|x-x'\| < \delta$, se verifica que $\|f(x)-f(x')\| < \varepsilon$.
\end{definicion}

Es obvio que $f$ uniformemente continua $\implies f$ continua (en todos los puntos de $X$). El recíproco es falso en general, pero es cierto si $X$ es compacto.

\begin{proposicion}{Lema (compacto $\Rightarrow$ uniformemente continua) [C8.10] \marcaL}{lema_unif_continua}
Sea $f: X \subseteq \mathbb{R}^n \to \mathbb{R}^m$ continua, con $X$ compacto $\implies f$ uniformemente continua.
\end{proposicion}

\begin{proof}
Sea $\varepsilon > 0$. Sea $x \in X$. Por hipótesis:
$\exists \delta_x > 0$ tal que $\forall x' \in X$ con $\|x-x'\| < \delta_x$, se cumple que $\|f(x)-f(x')\| < \varepsilon/2$. ($B(x,\delta_x) = \{y \in X \mid \|x-y\| < \delta_x\}$, bola abierta de centro $x$, radio $\delta_x$).
Claramente, $X = \bigcup_{x\in X} B(x, \delta_x/2)$.
Como $X$ es compacto, admite un subrecubrimiento finito: $\exists x_1, \dots, x_r \in X$ tales que $X = \bigcup_{j=1}^r B(x_j, \delta_{x_j}/2)$.
Sea $\delta = \min \left\{\frac{\delta_{x_1}}{2}, \dots, \frac{\delta_{x_r}}{2}\right\}$.
Veamos que $\delta$ funciona:
Sea $x,x' \in X$ con $\|x-x'\| < \delta$. Veamos que $\|f(x)-f(x')\| < \varepsilon$.
Dado que $x \in X$, $\exists j \in \{1,\dots,r\}$ tal que $x \in B(x_j, \delta_{x_j}/2)$, por tanto, $\|x-x_j\| < \delta_{x_j}/2$.
Observamos que:
\[ \|x'-x_j\| \leq \|x'-x\| + \|x-x_j\| < \delta + \frac{\delta_{x_j}}{2} \leq \frac{\delta_{x_j}}{2} + \frac{\delta_{x_j}}{2} = \delta_{x_j} \implies \|f(x')-f(x_j)\| < \frac{\varepsilon}{2} \]
Finalmente:
\[ \|f(x)-f(x')\| \leq \|f(x)-f(x_j)\| + \|f(x_j)-f(x')\| < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon. \]
\end{proof}

\begin{teorema}{Existencia de elevaciones [C8.11] \marcaT}{teo_existencia_elevaciones}
Sea $X \subseteq \mathbb{R}^n$. Supongamos que $X$ es convexo y compacto. Sea $f: X \to S^1$ continua. Sea $x_0 \in X$. Sea $t_0 \in \mathbb{R}$ tal que $p(t_0) = f(x_0)$. Entonces existe una única elevación $\tilde{f} : X \to \mathbb{R}$ de $f$ tal que $\tilde{f}(x_0) = t_0$.
\end{teorema}

\begin{proof}
\textbf{Unicidad:} por la \cref{prop:prop_unicidad_elev}, ya que $X$ es convexo y, por tanto, conexo.

\textbf{Existencia con $x_0 = 0$.}
\begin{itemize}
    \item $X$ es compacto, luego $f$ es uniformemente continua (\cref{prop:lema_unif_continua}): existe $\delta > 0$ con $\|x - x'\| < \delta \Rightarrow \|f(x) - f(x')\| < 2$. Además $X$ está acotado, $\|x\| \leq K$, y tomamos $m \in \mathbb{Z}_{\geq 1}$ con $\frac{K}{\delta} < m$.
    \item Dados $j \in \{0,\dots,m-1\}$ y $x \in X$, los puntos $\frac jm x$ y $\frac{j+1}{m}x$ están en $X$ ($X$ es convexo y $0 \in X$) y distan $\frac{\|x\|}{m} < \delta$. Luego sus imágenes por $f$ no son opuestas y
    \[ g_j \colon X \to S^1 \setminus \{(-1,0)\}, \qquad g_j(x) = \frac{f\left(\frac{j+1}{m}x\right)}{f\left(\frac jm x\right)} \]
    está bien definida y es continua.
    \item Con $\Psi$ la inversa del \cref{prop:lema_restriccion_p}, definimos la aplicación continua
    \[ \tilde f(x) = t_0 + \sum_{j=0}^{m-1} \Psi\big(g_j(x)\big). \]
    Como $g_j(0) = 1$, $\tilde f(0) = t_0$. Como $p$ es homomorfismo (\cref{obs:obs_p_homomorfismo}) y $p(t_0) = f(x_0) = f(0)$, el producto es telescópico:
    \[ p(\tilde f(x)) = f(0) \cdot \frac{f(\frac1m x)}{f(0)} \cdot \frac{f(\frac2m x)}{f(\frac1m x)} \cdots \frac{f(x)}{f(\frac{m-1}{m}x)} = f(x). \]
\end{itemize}

\textbf{Caso general.} Sea $L(x) = x - x_0$, una traslación, luego un homeomorfismo de $\mathbb{R}^n$ con inversa $L^{-1}(y) = y + x_0$. $L(X)$ es convexo, compacto y contiene a $0 = L(x_0)$. Aplicando lo anterior a $g = f \circ L^{-1}$ se obtiene $\tilde g$ con $\tilde g(0) = t_0$ y $p \circ \tilde g = g$, y $\tilde f = \tilde g \circ L$ cumple $p \circ \tilde f = g \circ L = f$ y $\tilde f(x_0) = \tilde g(0) = t_0$.
\end{proof}

La recta $\mathbb{R}$ se enrolla sobre $S^1$ como una hélice; $p$ la ``mira desde arriba'' y olvida en qué vuelta estamos (todos los enteros caen en $(1,0)$). En la figura, un camino $\alpha$ que da $1{,}4$ vueltas y su elevación $\tilde\alpha$, que sube de $0$ a $1{,}4$.
\begin{itemize}
    \item \textbf{Elevación} (\cref{def:def_elevacion}): $\tilde f \colon X \to \mathbb{R}$ con $p \circ \tilde f = f$. Es elegir, de forma continua, en qué vuelta de la hélice está cada $f(x)$: el ángulo de $f(x)$ sin reducir módulo $1$.
    \item \textbf{Unicidad} (\cref{prop:prop_unicidad_elev}): si $X$ es conexo, dos elevaciones que coinciden en un punto son iguales. En general, dos elevaciones difieren en una constante entera (cambiar de vuelta).
    \item \textbf{Existencia} (\cref{thm:teo_existencia_elevaciones}): si $X$ es convexo y compacto, para cada $t_0$ con $p(t_0) = f(x_0)$ hay una única elevación con $\tilde f(x_0) = t_0$.
    \item \textbf{Lo que usaremos:} $I$ e $I \times I$ son convexos y compactos, así que se elevan los arcos y las homotopías. Si $\alpha$ es un lazo en $(1,0)$ y $\tilde\alpha(0) = 0$, entonces $\tilde\alpha(1) \in p^{-1}(1,0) = \mathbb{Z}$: el número de vueltas, que será el grado.
\end{itemize}
\end{nota}

\begin{proposicion}{Elevación de arcos [C9.1] \marcaC}{cor_elev_arcos}
Sea $\alpha \colon I \to S^1$ continua tal que $\alpha(0) = (1,0)$. Entonces existe una única elevación $\tilde{\alpha} \colon I \to \mathbb{R}$ de $\alpha$ tal que $\tilde{\alpha}(0) = 0$.
\end{proposicion}

\begin{proposicion}{Elevación de homotopías [C9.2] \marcaC}{cor_elev_homotopias}
Sea $F \colon I \times I \to S^1$ continua tal que $F(0,0) = (1,0)$. Entonces existe una única elevación $\tilde{F} \colon I \times I \to \mathbb{R}$ de $F$ tal que $\tilde{F}(0,0) = 0$.
\end{proposicion}

Ambos resultados se deducen del \cref{thm:teo_existencia_elevaciones}, ya que $I$ e $I \times I$ son convexos y compactos.

\begin{definicion}{Grado de un lazo [C9.3]}{def_grado}
Sea $\alpha \in C_{(1,0)}(S^1)$. Se define el \textbf{grado} de $\alpha$ como $\deg(\alpha) := \tilde{\alpha}(1)$, siendo $\tilde{\alpha}$ la única elevación de $\alpha$ con $\tilde{\alpha}(0) = 0$. Está bien definido por la elevación de arcos.
\end{definicion}

Geométricamente, $\deg(\alpha)$ es el número de vueltas completas de $\alpha$ en sentido antihorario. Como $p(\tilde{\alpha}(1)) = \alpha(1) = (1,0)$, se tiene $\tilde{\alpha}(1) \in p^{-1}(1,0) = \mathbb{Z}$, luego $\deg(\alpha) \in \mathbb{Z}$.
Solo esta definido si $\alpha$ empieza y acaba en $(1,0)$, lo que significa que $\alpha$ solo da vueltas completas, por lo que las podemos contar.

\textbf{Ejemplo.} Para $n \in \mathbb{Z}$, sea $\omega_n \colon I \to S^1$, $\omega_n(t) = e^{2\pi i n t}$. Su elevación con $\tilde{\omega}_n(0)=0$ es $\tilde{\omega}_n(t) = nt$, luego $\deg(\omega_n) = n$.

La aplicación $\deg \colon \pi_1(S^1,(1,0)) \to \mathbb{Z}$, $[\alpha] \mapsto \deg(\alpha)$, es un isomorfismo de grupos.

    \draw[-{Stealth[length=6pt]}, thick, gray!60!black] (2.4,0.3) to[bend left=15] node[above, font=\small] {$\tilde F$} (4.2,1.3);
    \draw[-{Stealth[length=6pt]}, thick, gray!60!black] (2.4,-1.4) to[bend right=15] node[below left, font=\small] {$F$} (5.4,-2.85);
    \draw[-{Stealth[length=6pt]}, thick, gray!60!black] (9.9,1.0) to[bend left=20] node[right, font=\small] {$p$} (9.2,-2.6);
    \end{tikzpicture}
    \end{center}
    Como $F(0,t) = F(1,t) = (1,0)$, al elevar cada lado cae en $p^{-1}(1,0) = \mathbb{Z}$ y, por conexión, es constante: el izquierdo vale $0$ y el derecho $k$. Ese $\tilde F(1,t) = k$ es el número de vueltas, y como no depende de $t$, es el mismo para $\alpha$ ($t=0$) y para $\beta$ ($t=1$).

    \item \textbf{Es homomorfismo.} Sean $\alpha, \beta \in C_{(1,0)}(S^1)$ con elevaciones $\tilde{\alpha}, \tilde{\beta}$ que valen $0$ en $0$. Definimos $w \colon I \to \mathbb{R}$ por
    \[ w(s) = \begin{cases} \tilde{\alpha}(2s), & s \in [0,\tfrac12], \\ \tilde{\alpha}(1) + \tilde{\beta}(2s-1), & s \in [\tfrac12,1]. \end{cases} \]
    Es continua (en $s=\tfrac12$ ambas valen $\tilde{\alpha}(1)$), $w(0)=0$ y $p \circ w = \alpha * \beta$, pues $p(\tilde{\alpha}(1) + \tilde{\beta}(u)) = p(\tilde{\alpha}(1))\,p(\tilde{\beta}(u)) = \beta(u)$ ya que $p(\tilde{\alpha}(1)) = (1,0)$. Luego $w$ es la elevación de $\alpha*\beta$ que empieza en $0$ y
    \[ \deg(\alpha * \beta) = w(1) = \tilde{\alpha}(1) + \tilde{\beta}(1) = \deg(\alpha) + \deg(\beta). \]

    \item \textbf{Inyectiva.} Sean $\alpha, \beta \in C_{(1,0)}(S^1)$ con $\deg(\alpha) = \deg(\beta)$. Consideramos la homotopía convexa entre sus elevaciones
    \[ H \colon I \times I \to \mathbb{R}, \quad H(s,t) = (1-t)\tilde{\alpha}(s) + t\tilde{\beta}(s). \]
    Entonces $H(0,t) = 0$ y $H(1,t) = (1-t)\deg(\alpha) + t\deg(\beta) = \deg(\alpha)$ no dependen de $t$, es decir, $H \colon \tilde{\alpha} \simeq_{\{0,1\}} \tilde{\beta}$. Sea $F = p \circ H$. Entonces $F_0 = p\circ\tilde{\alpha} = \alpha$, $F_1 = \beta$, $F(0,t) = p(0) = (1,0)$ y $F(1,t) = p(\deg\alpha) = (1,0)$, luego $F \colon \alpha \simeq_{\{0,1\}} \beta$.

    \item \textbf{Sobreyectiva.} Dado $n \in \mathbb{Z}$, el lazo $\omega_n$ cumple $\deg([\omega_n]) = n$ (ejemplo anterior).
\end{enumerate}
\end{proof}

\begin{proposicion}{$S^1$ no es contráctil [C9.6] \marcaC}{cor_s1_no_contractil}
$S^1$ no es contráctil, ya que $\pi_1(S^1) \cong \mathbb{Z} \neq 1$.
\end{proposicion}

\begin{proof}
Supongamos que $S^1$ es contráctil. Por la caracterización de contráctiles (\cref{thm:teo_contractil}), $S^1$ tiene el mismo tipo de homotopía que un espacio $\{p\}$ formado por un único punto. Como $S^1$ y $\{p\}$ son arco-conexos y el grupo fundamental es un invariante homotópico (\cref{prop:equiv_homotopica}), se tiene $\pi_1(S^1) \cong \pi_1(\{p\}) = 1$, es decir, $\mathbb{Z} \cong 1$, lo cual es absurdo. Luego $S^1$ no es contráctil.
\end{proof}

\begin{proposicion}{$S^2$ y $T^2$ no son homeomorfos [C9.7] \marcaC}{cor_s2_t2}
$S^2$ y $T^2$ no son homeomorfos.
\end{proposicion}

\begin{proof}
$S^2$ es simplemente conexo, así que $\pi_1(S^2) = 1$, mientras que $\pi_1(T^2) \cong \mathbb{Z} \times \mathbb{Z}$ (\cref{prop:ej_pi1_producto} con $T^2 = S^1 \times S^1$). Como el grupo fundamental es un invariante topológico, $S^2 \not\cong T^2$.
\end{proof}

\begin{proposicion}{Grupo fundamental de un producto [C9.8] \marcaE}{ej_pi1_producto}
Sean $X$ e $Y$ espacios topológicos y $(x_0,y_0) \in X \times Y$. Entonces
\[ \pi_1(X \times Y, (x_0,y_0)) \cong \pi_1(X,x_0) \times \pi_1(Y,y_0). \]
En particular, si $X$ e $Y$ son arco-conexos, $\pi_1(X \times Y) \cong \pi_1(X) \times \pi_1(Y)$.
\end{proposicion}

\begin{proof}
Sean $p_X \colon X \times Y \to X$ y $p_Y \colon X \times Y \to Y$ las proyecciones. Definimos
\[ \varphi \colon \pi_1(X \times Y,(x_0,y_0)) \to \pi_1(X,x_0) \times \pi_1(Y,y_0), \qquad \varphi([\gamma]) = \big((p_X)_*[\gamma],\, (p_Y)_*[\gamma]\big) = \big([p_X \circ \gamma],\, [p_Y \circ \gamma]\big). \]
Está bien definida y es un homomorfismo, porque sus dos componentes son homomorfismos inducidos (\cref{thm:teo_hom_ind}).

Para la inversa, dados $\alpha \in C_{x_0}(X)$ y $\beta \in C_{y_0}(Y)$, sea $(\alpha,\beta)(s) = (\alpha(s),\beta(s))$, que es un lazo en $(x_0,y_0)$, y sea $\psi([\alpha],[\beta]) = [(\alpha,\beta)]$. Está bien definida: si $H_1 \colon \alpha \simeq_{\{0,1\}} \alpha'$ y $H_2 \colon \beta \simeq_{\{0,1\}} \beta'$, entonces $(H_1,H_2) \colon (\alpha,\beta) \simeq_{\{0,1\}} (\alpha',\beta')$, que es continua por serlo sus componentes y fija los extremos.

Se tiene $\varphi \circ \psi = 1$, porque $p_X \circ (\alpha,\beta) = \alpha$ y $p_Y \circ (\alpha,\beta) = \beta$; y $\psi \circ \varphi = 1$, porque $\gamma = (p_X \circ \gamma,\, p_Y \circ \gamma)$. Luego $\varphi$ es un homomorfismo biyectivo, es decir, un isomorfismo. Si $X$ e $Y$ son arco-conexos, también lo es $X \times Y$, y no depende del punto base (\cref{prop:cor_arco}).
\end{proof}

\begin{observacion}{Grupo fundamental del cilindro [C9.9] \marcaC}{obs_cilindro}
El grupo fundamental del cilindro $S^1 \times I$ es $\mathbb{Z}$, ya que $S^1$ y $S^1 \times I$ son del mismo tipo de homotopía.
\end{observacion}
