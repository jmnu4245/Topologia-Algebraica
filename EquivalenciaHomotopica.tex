\section{Equivalencia homotópica}

\subsection{Motivación: Homeomorfismo vs. Homotopía}

Sean $X, Y$ espacios topológicos. Decimos que $X$ e $Y$ son homeomorfos cuando $\exists f \colon X \to Y$ homeomorfismo ($f$ continua, biyectiva, $f^{-1}$ continua).

Equivalentemente: $X \cong Y \iff \exists f \colon X \to Y$ continua, $\exists g \colon Y \to X$ continua, tales que $f \circ g = 1_Y$ y $g \circ f = 1_X$.

\begin{nota}{Notación}{nota_homeomorfismo}
$X \cong Y$ significa que $X$ e $Y$ son homeomorfos. La aplicación identidad en $X$ se denota $1_X \colon X \to X$, $x \mapsto x$ (también se usa $id_X, i_X$).
\end{nota}

En este tema relajaremos las condiciones de la definición de homeomorfismo para llegar a la noción de espacios del mismo tipo de homotopía. Buscamos definir una especie de forma fundamental constante ante contracciones (como ilustra la reducción de un cilindro a una circunferencia o de un cono a un vértice).

\subsection{Homotopía entre aplicaciones}

\begin{definicion}{Homotopía entre aplicaciones [C2.2]}{def_homotopia}
Sea $f \colon X \to Y$ aplicación continua. \\
Sea $g \colon X \to Y$ aplicación continua.

Diremos que $f, g$ son homotópicas cuando $\exists F \colon X \times I = [0,1] \to Y$ continua tal que $F_0 = f, F_1 = g$.
\end{definicion}

\begin{nota}{Notación [C2.1]}{nota_homotopia}
Sea $t \in [0,1]$, definimos $F_t \colon X \to Y$ como $x \mapsto F(x,t)$.
\end{nota}

$F$ captura una transformación parametrizada y continua entre $f$ y $g$.

\vspace{0.5cm}
\textbf{Ejemplos:} ¿$f$ y $g$ son homotópicas?

Denotamos por $S^n = \{x \in \mathbb{R}^{n+1} \mid |x| = 1\}$ la esfera de dimensión $n$ (así, $S^0 = \{-1,1\}$ y $S^1$ es la circunferencia).

\begin{enumerate}
    \item $f \colon S^1 \to S^1, \quad z \mapsto z$ \\
    $g \colon S^1 \to S^1, \quad z \mapsto -z$
    
    $F \colon S^1 \times I \to S^1$ \\
    $(z, t) \mapsto z(\cos(\pi t) + i \sin(\pi t))$. 
    
    Como $F$ es continua y $F_0 = f, F_1 = g$, concluimos que $f$ y $g$ son homotópicas.
    
    \item $f \colon \mathbb{R}^2 \to \mathbb{R}^2, \quad (x,y) \mapsto (x,0)$ \\
    $g \colon \mathbb{R}^2 \to \mathbb{R}^2, \quad (x,y) \mapsto (0,y)$
    
    $F \colon \mathbb{R}^2 \times I \to \mathbb{R}^2$ \\
    $((x,y), t) \mapsto ((1-t)x, ty)$
    
    Por tanto $f$ y $g$ son homotópicas ($f \simeq g$).
\end{enumerate}

\subsubsection{Estructura en $C(X,Y)$}

Llamamos $C(X,Y) = \{f \colon X \to Y \mid f \text{ continua}\}$ al conjunto de aplicaciones continuas entre $X$ e $Y$.

\begin{teorema}{Relación de equivalencia en $C(X,Y)$ [C2.5]}{teo_rel_eq}
La homotopía $\simeq$ es una Relación de Equivalencia (R.B.E.) en $C(X,Y)$.
\end{teorema}

\begin{proof}
\begin{enumerate}
    \item \textbf{Reflexiva:} $\forall f \in C(X,Y) \colon f \simeq f$. Basta definir $F \colon X \times I \to Y$ continua mediante $(x,t) \mapsto f(x)$. Evaluando, $F_0 = f$ y $F_1 = f$.
    \item \textbf{Simétrica:} $\forall f,g \in C(X,Y) \colon f \simeq g \implies g \simeq f$.
    Dado que $f \simeq g$, $\exists F \colon X \times I \to Y$ continua tal que $F_0 = f, F_1 = g$.
    Buscamos una $G \colon X \times I \to Y$ continua con $G_0 = g, G_1 = f$.
    Definimos $G(x,t) = F(x, 1-t)$. Es continua por ser composición de funciones continuas y cumple $G_0 = F_1 = g$ y $G_1 = F_0 = f$.
    \item \textbf{Transitiva:} $\forall f,g,h \in C(X,Y) \colon f \simeq g, g \simeq h \implies f \simeq h$.
    Suponemos que $f \simeq^{(1)} g$ y $g \simeq^{(2)} h$. Esto implica que existen $F^{(1)} \colon X \times [0,1] \to Y$ y $F^{(2)} \colon X \times [0,1] \to Y$.
    Necesitamos construir $F \colon X \times I \to Y$ continua con $F_0 = f$ y $F_1 = h$. Definimos:
    \[ F(x,t) = \begin{cases} F^{(1)}(x, 2t) & \forall t \in \left[0, \frac{1}{2}\right] \\ F^{(2)}(x, 2t-1) & \forall t \in \left[\frac{1}{2}, 1\right] \end{cases} \]
    Como $\left(X \times \left[0,\frac{1}{2}\right]\right)$ y $\left(X \times \left[\frac{1}{2}, 1\right]\right)$ son cerrados con restricción continua en la intersección, entonces $F$ es continua. Comprobamos la frontera:
    $F\left(x, \frac{1}{2}\right) = F^{(1)}(x,1) = g(x)$ y $F\left(x, \frac{1}{2}\right) = F^{(2)}(x,0) = g(x)$. Tenemos $F$ bien definida.
    $F_0(x) = F(x,0) = F^{(1)}(x,0) = f(x), \forall x \in X$.
    $F_1(x) = F(x,1) = F^{(2)}(x,1) = h(x), \forall x \in X$.
    Como consecuencia de todo lo anterior, $f \simeq h$.
\end{enumerate}
\end{proof}

\begin{definicion}{Conjunto de clases de homotopía [C3.2]}{def_cociente_homotopia}
Como $\simeq$ es una relación de equivalencia en $C(X,Y)$ (\cref{thm:teo_rel_eq}), podemos considerar el conjunto cociente $C(X,Y)/\simeq$, que se denota por $[X,Y]$.
\end{definicion}

\begin{teorema}{Composición de homotopías [C2.6]}{teo_comp_homo}
Sean $X \xrightarrow{f_1, f_2} Y \xrightarrow{g_1, g_2} Z$. Si suponemos que $f_1 \simeq f_2$ y $g_1 \simeq g_2$, entonces $g_1 \circ f_1 \simeq g_2 \circ f_2$.
\end{teorema}

\begin{proof}
Tenemos $F \colon X \times I \to Y$ continua con $F_0=f_1, F_1=f_2$.
Tenemos $G \colon Y \times I \to Z$ continua con $G_0=g_1, G_1=g_2$.
Definimos la aplicación $H \colon X \times I \to Z$, $(x,t) \mapsto G(F(x,t), t)$. Evaluamos en los extremos:
\[ H_0(x) = H(x,0) = G(F(x,0), 0) = G(f_1(x), 0) = G_0(f_1(x)) = g_1(f_1(x)), \forall x \in X. \]
\[ H_1(x) = H(x,1) = G(F(x,1), 1) = G(f_2(x), 1) = G_1(f_2(x)) = g_2(f_2(x)), \forall x \in X. \]
Como $H$ es continua, queda probado que $g_1 \circ f_1 \simeq g_2 \circ f_2$.
\end{proof}


\subsection{Equivalencia homotópica entre espacios}

\begin{definicion}{Equivalencia homotópica [C2.7, C3.1]}{def_eq_homotopica}
Sean $X, Y$ espacios topológicos. Diremos que $X$ e $Y$ son del mismo tipo de homotopía ($X \simeq Y$) cuando $\exists f \colon X \to Y$ continua y $\exists g \colon Y \to X$ continua tales que $f \circ g \simeq 1_Y$ y $g \circ f \simeq 1_X$. 

En este caso también diremos que $f$ es una \textbf{equivalencia homotópica} y $g$ la \textbf{inversa homotópica} de $f$ (no necesariamente $f$ biyectiva).
\end{definicion}

\begin{observacion}{Relación con el homeomorfismo [C2.8]}{obs_iso}
$X \cong Y \implies X \simeq Y$ (y $g$ es inversa de $f$).
Tomando $f \colon X \to Y, g \colon Y \to X$ biyecciones donde $g \circ f = 1_X$ y $f \circ g = 1_Y$, se sigue que son trivialmente homotópicas a las identidades ($g \circ f \simeq 1_X$ y $f \circ g \simeq 1_Y$).

Sin embargo, $f \colon X \to Y$ continua y biyectiva $\not\implies f^{-1}$ continua (y no necesariamente $f$ es equivalencia homotópica si falla el homeomorfismo). Como contraejemplo:
$f \colon [0, 1) \to S^1, \quad t \mapsto (\cos(2\pi t), \sin(2\pi t))$.
\end{observacion}

\textbf{Ejemplo ($[0,1] \simeq \{0\}$ pero $[0,1] \not\cong \{0\}$).}
Sean $X = [0,1]$ e $Y = \{0\}$. Definimos $f \colon X \to Y$, $f(x) = 0$, y $g \colon Y \to X$, $g(0) = 0$. Entonces $f \circ g = 1_Y$, y $g \circ f = c_0$ es homotópica a $1_X$ mediante $F \colon X \times I \to X$, $F(x,t) = tx$ (continua, con $F_0 = c_0$ y $F_1 = 1_X$). Por tanto $[0,1] \simeq \{0\}$.

Sin embargo $[0,1] \not\cong \{0\}$, pues no existe ninguna biyección entre un conjunto con un único elemento e $[0,1]$. Así, ``tener el mismo tipo de homotopía'' es estrictamente más débil que ``ser homeomorfos''.

\textbf{Ejemplos:}
\begin{enumerate}
    \item Sea $X = D(0,1) = \{(x,y) \in \mathbb{R}^2 \mid ||(x,y)|| < 1\}$ y $Y = \{(0,0)\}$.
    Como $D(0,1)$ es convexo, es contráctil (\cref{thm:teo_convexo_contractil}), es decir, del mismo tipo de homotopía que un punto (\cref{thm:teo_contractil}).
    
    \item Sean $X = [-1,1]$ y $Y = [-1,1] \times \{0\} \cup \{0\} \times [-1,1]$. 
    Por la \cref{prop:Theta_invariante}, como $\Theta(X) = \{1,2\} \neq \{1,2,4\} = \Theta(Y)$, se tiene $X \not\cong Y$. Sin embargo, se puede demostrar que $X \simeq Y$.
    
    \item Sean $X = S^1 \times [-1,1]$ (cilindro) y $Y = S^1$.
    $f \colon X \to Y, (x,y,z) \mapsto (x,y)$.
    $g \colon Y \to X, (x,y) \mapsto (x,y,0)$.
    
    $f \circ g(x,y) = f(x,y,0) = (x,y) = id_Y, \forall (x,y) \in Y$.
    $g \circ f(x,y,z) = g(x,y) = (x,y,0)$. 
    Construimos $F \colon X \times I \to X$ dada por $F((x,y,z), t) = (x,y, t \cdot z)$. Se verifica que $F_0 = g \circ f$ y $F_1 = 1_X \implies X \simeq Y$.
\end{enumerate}

\begin{nota}{Extraer un punto: caso topológico y caso homotópico}{nota_extraer_punto}
Si $f \colon X \to Y$ es un homeomorfismo y $P \in X$, entonces $f$ restringe a un homeomorfismo $X \setminus \{P\} \cong Y \setminus \{f(P)\}$ (es lo que se usa en la \cref{prop:Theta_invariante}).

Para el caso homotópico, la pregunta análoga es si, cuando $X \simeq Y$, se tiene $\forall P \in X,\ \exists Q \in Y$ tal que $X \setminus \{P\} \simeq Y \setminus \{Q\}$. La respuesta es \textbf{negativa}: el cilindro $Y = S^1 \times [-1,1]$ es del mismo tipo de homotopía que $X = S^1$ (ejemplo anterior), pero $S^1 \setminus \{P\}$ es homeomorfo a un intervalo, luego contráctil, mientras que $Y \setminus \{Q\}$ no lo es para ningún $Q$ (su grupo fundamental no es trivial).
\end{nota}

\begin{definicion}{Unión por un punto y ramo de circunferencias [C2.9]}{def_wedge}
Sean $(X,x_0)$ e $(Y,y_0)$ espacios con un punto base. Su \textbf{unión por un punto} es
\[ X \vee Y = (X \sqcup Y)/(x_0 \sim y_0), \]
es decir, se pegan $X$ e $Y$ identificando $x_0$ con $y_0$. Escribimos $\bigvee_k S^1 = S^1 \vee \dots \vee S^1$ ($k$ veces) y lo llamamos \textbf{ramo de $k$ circunferencias}: $k$ circunferencias que comparten un único punto. Para $k=2$ es la ``figura ocho'' $S^1 \vee S^1$; por convenio $\bigvee_0 S^1$ es un punto.
\end{definicion}

Más ejemplos de espacios del mismo tipo de homotopía:
\begin{enumerate}
    \setcounter{enumi}{3}
    \item $S^2 \setminus \{N\} \simeq \{\text{punto}\}$, con $N = (0,0,1)$.

    La proyección estereográfica es un homeomorfismo $S^2 \setminus \{N\} \cong \mathbb{R}^2$ (véase la \cref{prop:proy_estereografica}, que lo prueba para $S^n$). Como $\mathbb{R}^2$ es convexo, es contráctil; y como homeomorfo implica mismo tipo de homotopía y $\simeq$ es transitiva, $S^2 \setminus \{N\} \simeq \mathbb{R}^2 \simeq \{\text{punto}\}$.

    \item $\mathbb{R}^2 \setminus \{(0,0)\} \simeq S^1$.

    Sean $i \colon S^1 \hookrightarrow \mathbb{R}^2 \setminus \{0\}$ la inclusión y $r \colon \mathbb{R}^2 \setminus \{0\} \to S^1$, $r(x) = x/|x|$. Ambas son continuas y $r \circ i = 1_{S^1}$. Definimos
    \[ F \colon (\mathbb{R}^2 \setminus \{0\}) \times I \to \mathbb{R}^2 \setminus \{0\}, \qquad F(x,t) = (1-t)\,x + t\,\frac{x}{|x|} = \left( (1-t) + \frac{t}{|x|} \right) x. \]
    El coeficiente de $x$ es positivo, luego $F(x,t) \neq 0$ y $F$ está bien definida y es continua. Además $F_0 = 1_{\mathbb{R}^2 \setminus \{0\}}$ y $F_1 = i \circ r$. Por tanto $i \circ r \simeq 1$ y $\mathbb{R}^2 \setminus \{0\} \simeq S^1$.

    \item $\mathbb{R}^2 \setminus \{p_1, \dots, p_k\} \simeq \bigvee_k S^1$ (con $p_1, \dots, p_k$ distintos).

    \emph{Idea} (no se hace una demostración formal): se generaliza el ejemplo anterior. Alrededor de cada $p_i$ se contrae radialmente hasta una circunferencia, y el resto del plano se deforma sobre la unión de estas circunferencias, que quedan unidas en un único punto.

    \item $S^2 \setminus \{p_1, \dots, p_k\} \simeq \bigvee_{k-1} S^1$ (con $p_1, \dots, p_k$ distintos, $k \ge 1$).

    La proyección estereográfica desde $p_k$ da un homeomorfismo $S^2 \setminus \{p_k\} \cong \mathbb{R}^2$, que lleva $p_1, \dots, p_{k-1}$ a $k-1$ puntos distintos de $\mathbb{R}^2$. Por tanto $S^2 \setminus \{p_1, \dots, p_k\} \cong \mathbb{R}^2 \setminus \{q_1, \dots, q_{k-1}\} \simeq \bigvee_{k-1} S^1$ por el ejemplo anterior (para $k=1$ se recupera el ejemplo 4).

    \item $T^2 \setminus \{p\} \simeq S^1 \vee S^1$, siendo $T^2 = S^1 \times S^1$ el toro.

    \emph{Idea} (no se hace una demostración formal): se representa $T^2 = [0,1]^2/\!\sim$ identificando lados opuestos, con $p$ el centro del cuadrado. La proyección radial desde $p$ hacia el borde del cuadrado es compatible con las identificaciones y deforma $T^2 \setminus \{p\}$ sobre la imagen del borde, que son dos circunferencias (los lados identificados) con un único vértice común, es decir $S^1 \vee S^1$.
\end{enumerate}


\subsection{Espacios contráctiles}

\begin{definicion}{Espacio topológico contractil [C2.11]}{def_contractil}
Un espacio topológico $X \neq \emptyset$ es contractil si $\exists x_0 \in X \mid 1_X \simeq c_{x_0}$ (donde $c_{x_0}$ es la aplicación constante $x \mapsto x_0$).
\end{definicion}

\begin{teorema}{Caracterización de espacios contractiles [C2.12]}{teo_contractil}
Son equivalentes las siguientes afirmaciones para un e.t. $X \neq \emptyset$:
\begin{enumerate}
    \item[a)] $X$ es contractil.
    \item[b)] $id_X \simeq c_x, \forall x \in X$.
    \item[c)] $\forall f,g \colon T \to X$ continuas $\implies f \simeq g$ (para cualquier e.t. $T$).
    \item[d)] $X$ es del mismo tipo de homotopía que un punto.
\end{enumerate}
\end{teorema}

\begin{proof}
Probamos (a)$\Rightarrow$(b)$\Rightarrow$(c)$\Rightarrow$(a) y (a)$\Leftrightarrow$(d).

(a)$\Rightarrow$(b): Sea $x_0 \in X$ con $1_X \simeq c_{x_0}$. Dado $x \in X$, como $X$ es arco-conexo (por ser contráctil, \cref{thm:teo_contractil_arcoconexo}, cuya prueba no usa este resultado) existe un arco $\omega \colon I \to X$ con $\omega(0) = x_0$ y $\omega(1) = x$. La aplicación $H \colon X \times I \to X$, $H(y,t) = \omega(t)$, es continua con $H_0 = c_{x_0}$ y $H_1 = c_x$, luego $c_{x_0} \simeq c_x$. Por transitividad, $1_X \simeq c_{x_0} \simeq c_x$.

(b)$\Rightarrow$(c): Sean $f, g \colon T \to X$ continuas y fijemos $x \in X$. Como $1_X \simeq c_x$ y $f \simeq f$, el teorema de composición de homotopías da $1_X \circ f \simeq c_x \circ f$, es decir, $f \simeq c_x$ (donde ahora $c_x \colon T \to X$). Igualmente $g \simeq c_x$. Por simetría y transitividad, $f \simeq g$.

(c)$\Rightarrow$(a): Tomando $T = X$, $f = 1_X$ y $g = c_{x_0}$ (con $x_0 \in X$ cualquiera, pues $X \neq \emptyset$) obtenemos $1_X \simeq c_{x_0}$.

(a)$\Rightarrow$(d): Sea $x_0 \in X$ con $1_X \simeq c_{x_0}$. Definimos $f \colon X \to \{x_0\}$ constante y $g \colon \{x_0\} \to X$ la inclusión. Entonces $f \circ g = 1_{\{x_0\}}$ y $g \circ f = c_{x_0} \simeq 1_X$.

(d)$\Rightarrow$(a): Sean $f \colon X \to \{p\}$ y $g \colon \{p\} \to X$ con $g \circ f \simeq 1_X$. Pero $g \circ f = c_{g(p)}$, luego $1_X \simeq c_{x_0}$ con $x_0 = g(p)$.
\end{proof}

\subsection{Conjuntos convexos y estrellados}

\begin{definicion}{Conjunto convexo [C3.3]}{def_convexo}
Sea $X \subseteq \mathbb{R}^n$. Diremos que $X$ es convexo cuando todo segmento que une dos puntos cualesquiera de $X$ está contenido en $X$. Equivalentemente:
\[ \forall x,y \in X, \forall t \in [0,1]: \quad tx + (1-t)y \in X \]
\end{definicion}

\begin{definicion}{Conjunto estrellado [C3.3]}{def_estrellado}
Sea $X \subseteq \mathbb{R}^n$. Diremos $X$ estrellado cuando $\exists x_0 \in X$ tal que todo segmento que une $x_0$ con algún punto de $X$ esté contenido en $X$. Equivalentemente:
\[ X \text{ estrellado} \iff \exists x_0 \in X \mid \forall x \in X, \forall t \in [0,1]: \quad tx + (1-t)x_0 \in X \]
\end{definicion}

\begin{observacion}{ [C3.3]}{obs_convexo_estrellado}
Todo conjunto convexo es estrellado (cualquier punto $x_0$ sirve como centro).
\end{observacion}

\begin{teorema}{Contracción de conjuntos estrellados [C3.3]}{teo_estrellado_contractil}
Sea $X \subseteq \mathbb{R}^n$. Si $X$ es estrellado, entonces $X$ es contráctil.
\end{teorema}

\begin{proof}
Sea $x_0$ el centro de $X$. Por ser $X$ estrellado, la aplicación $F \colon X \times I \to X$, $F(x,t) = tx + (1-t)x_0$, está bien definida (el segmento que une $x_0$ con $x$ está en $X$) y es continua. Además $F_0 = c_{x_0}$ y $F_1 = 1_X$, luego $1_X \simeq c_{x_0}$ y $X$ es contráctil.
\end{proof}

\begin{teorema}{Contracción de conjuntos convexos [C3.3]}{teo_convexo_contractil}
Todo subconjunto convexo de $\mathbb{R}^n$ es contráctil (del mismo tipo de homotopía que un punto).
\end{teorema}

\begin{proof}
Es consecuencia de la \cref{obs:obs_convexo_estrellado} y del \cref{thm:teo_estrellado_contractil}.
\end{proof}

\begin{observacion}{Los recíprocos no se cumplen [C3.3]}{obs_reciprocos}
Para subconjuntos de $\mathbb{R}^n$ tenemos convexo $\implies$ estrellado $\implies$ contráctil, pero ninguna de las dos implicaciones puede invertirse:
\begin{itemize}
    \item \textbf{Estrellado $\not\implies$ convexo.} La cruz $C = [-1,1] \times \{0\} \cup \{0\} \times [-1,1]$ es estrellada respecto del origen (el segmento que une $(0,0)$ con cualquier punto de $C$ está en $C$), pero no es convexa: $(1,0), (0,1) \in C$ y su punto medio $(\tfrac12, \tfrac12) \notin C$.
    \item \textbf{Contráctil $\not\implies$ estrellado.} Sea la ``U'' $U = \{0\} \times [0,1] \cup [0,1] \times \{0\} \cup \{1\} \times [0,1]$. Es homeomorfa a un intervalo, luego contráctil. Pero no es estrellada: dado $x_0 \in U$, si $x_0$ está en la base o en el lado izquierdo y $x_0 \neq (1,0)$ tomamos $y = (1,1)$; en otro caso tomamos $y = (0,1)$. En ambos casos los puntos interiores del segmento $[x_0, y]$ tienen abscisa en $]0,1[$ y ordenada $> 0$, y los únicos puntos de $U$ con ordenada $> 0$ tienen abscisa $0$ o $1$; luego $[x_0,y] \not\subseteq U$.
\end{itemize}
\end{observacion}
